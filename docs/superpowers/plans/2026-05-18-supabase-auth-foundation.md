# Lumen Auth + DB Foundation Implementation Plan (Plan 1 of 4)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Bring per-user authentication and Postgres-backed persistence to Lumen. After this plan lands, a signed-up user can use the existing `/app` UI exactly as before, but their journal data lives in Supabase under RLS, scoped to a per-user "active notebook" — the foundation that subsequent plans build dashboards and vault views on top of.

**Architecture:** Six coordinated changes. (1) Apply the schema from the spec as a single atomic Supabase migration. (2) Add `@supabase/ssr` browser + server clients and a `SessionProvider` mounted in `__root.tsx`. (3) New routes under `src/routes/auth/` (login, signup, callback). (4) `beforeLoad` gating on `/app` and a sign-out action wired into the existing welcome shell. (5) Flip both existing Supabase edge functions (`extract-graph`, `speak-agent`) to `verify_jwt = true` and scope their writes to the JWT's `auth.uid()` + the body's `notebook_id`. (6) Replace the localStorage backend inside `src/lib/journal-store.ts` with Supabase queries while keeping the exported function signatures unchanged, so the existing journal components don't need touching.

**Tech Stack:** TypeScript (strict), React 19, TanStack Start (SSR) + TanStack Router (file-based routing), `@supabase/ssr` (new), `@supabase/supabase-js` (already installed), Supabase Postgres + Edge Functions, Tailwind v4, shadcn/ui, Bun. No automated test harness — validation is `npx tsc --noEmit` plus manual smoke testing via `npm run dev` (matches the convention from the landing-redesign plan).

**Companion spec:** [`docs/superpowers/specs/2026-05-18-supabase-auth-and-vault-design.md`](../specs/2026-05-18-supabase-auth-and-vault-design.md)

**Branch:** `feat/supabase-auth-and-vault` (already created off `main`).

```ts
import { createServerClient, type CookieOptions } from "@supabase/ssr";
import type { Database } from "./types";

const SUPABASE_URL = import.meta.env.VITE_SUPABASE_URL as string;
const SUPABASE_KEY = import.meta.env.VITE_SUPABASE_PUBLISHABLE_KEY as string;

/**
 * Build a per-request Supabase client for SSR loaders. Cookies are
 * bridged via the provided `request` and `response` objects so the
 * session cookie set during sign-in is visible on the very next
 * server render.
 *
 * Usage inside a TanStack Start route loader:
 *   const supabase = getServerSupabase(request, response);
 *   const { data: { user } } = await supabase.auth.getUser();
 */
export function getServerSupabase(request: Request, response: Response) {
  const requestCookies = parseCookies(request.headers.get("cookie") ?? "");
  return createServerClient<Database>(SUPABASE_URL, SUPABASE_KEY, {
    cookies: {
      get(name: string) {
        return requestCookies[name];
      },
      set(name: string, value: string, options: CookieOptions) {
        response.headers.append("set-cookie", serializeCookie(name, value, options));
      },
      remove(name: string, options: CookieOptions) {
        response.headers.append(
          "set-cookie",
          serializeCookie(name, "", { ...options, maxAge: 0 }),
        );
      },
    },
  });
}

// ─── tiny cookie helpers (no extra deps) ──────────────────────────
function parseCookies(header: string): Record<string, string> {
  const out: Record<string, string> = {};
  for (const part of header.split(";")) {
    const trimmed = part.trim();
    if (!trimmed) continue;
    const eq = trimmed.indexOf("=");
    if (eq === -1) continue;
    const name = trimmed.slice(0, eq).trim();
    const value = decodeURIComponent(trimmed.slice(eq + 1).trim());
    out[name] = value;
  }
  return out;
}

function serializeCookie(name: string, value: string, options: CookieOptions): string {
  const parts: string[] = [`${name}=${encodeURIComponent(value)}`];
  if (options.maxAge !== undefined) parts.push(`Max-Age=${options.maxAge}`);
  if (options.domain) parts.push(`Domain=${options.domain}`);
  if (options.path) parts.push(`Path=${options.path}`);
  if (options.expires) parts.push(`Expires=${options.expires.toUTCString()}`);
  if (options.httpOnly) parts.push("HttpOnly");
  if (options.secure) parts.push("Secure");
  if (options.sameSite) parts.push(`SameSite=${options.sameSite}`);
  return parts.join("; ");
}
```

- [ ] **Step 2: Write the session module**

Create `src/lib/auth/session.ts` with:

```tsx
import { createContext, useContext, useEffect, useState, type ReactNode } from "react";
import type { Session, User, SupabaseClient } from "@supabase/supabase-js";
import { supabase } from "@/lib/supabase/browser";
import type { Database } from "@/lib/supabase/types";

type SessionContextValue = {
  session: Session | null;
  user: User | null;
  supabase: SupabaseClient<Database>;
};

const SessionContext = createContext<SessionContextValue | null>(null);

/**
 * Wrap the app once at the root. `initialSession` is whatever the SSR
 * loader saw — so the first paint already knows whether the user is
 * authenticated, avoiding a flash of signed-out UI.
 */
export function SessionProvider({
  initialSession,
  children,
}: {
  initialSession: Session | null;
  children: ReactNode;
}) {
  const [session, setSession] = useState<Session | null>(initialSession);

  useEffect(() => {
    const { data: { subscription } } = supabase.auth.onAuthStateChange((_event, next) => {
      setSession(next);
    });
    return () => {
      subscription.unsubscribe();
    };
  }, []);

  return (
    <SessionContext.Provider
      value={{ session, user: session?.user ?? null, supabase }}
    >
      {children}
    </SessionContext.Provider>
  );
}

export function useSession(): SessionContextValue {
  const ctx = useContext(SessionContext);
  if (!ctx) {
    throw new Error("useSession must be called inside <SessionProvider>");
  }
  return ctx;
}

export function useSupabase(): SupabaseClient<Database> {
  return useSession().supabase;
}
```

```ts
/**

export async function ensureActiveNotebook(userId: string): Promise<Notebook> {
  const { data: open } = await supabase
    .from("notebooks")
    .select("*")
    .eq("user_id", userId)
    .is("ended_at", null)
    .order("started_at", { ascending: false })
    .limit(1)

    .from("notebooks")

/**
 * Load the full JournalState for a given notebook. Identical shape
 * to the old localStorage version so consumers don't need changes.

export async function loadState(notebookId: string): Promise<JournalState> {
  const [entriesRes, entitiesRes, relationsRes, followUpsRes, askedRes] =
    await Promise.all([
      supabase
        .from("journal_entries")
        .select("*")
        .eq("notebook_id", notebookId)
        .order("created_at", { ascending: false }),
      supabase.from("entities").select("*").eq("notebook_id", notebookId),
      supabase.from("relations").select("*").eq("notebook_id", notebookId),
      supabase
        .from("follow_ups")
        .select("*")
        .eq("notebook_id", notebookId)
        .order("created_at", { ascending: false }),
      supabase
        .from("asked_questions")
        .select("*")
        .eq("notebook_id", notebookId)
        .order("asked_at", { ascending: false })
        .limit(ASKED_QUESTIONS_CAP),
    ]);

  if (entriesRes.error || entitiesRes.error || relationsRes.error
      || followUpsRes.error || askedRes.error) {
    console.warn("journal-store loadState partial failure", {
      entries: entriesRes.error, entities: entitiesRes.error,
      relations: relationsRes.error, followUps: followUpsRes.error,
      asked: askedRes.error,
    });
    return EMPTY_STATE;
  }

  return {
    entries: (entriesRes.data ?? []).map(rowToEntry),
    entities: Object.fromEntries(
      (entitiesRes.data ?? []).map((r) => [r.id, rowToEntity(r)]),
    ),
    relations: Object.fromEntries(
      (relationsRes.data ?? []).map((r) => [r.id, rowToRelation(r)]),

/**

 * the supplied state. Used by `mergeExtraction`'s consumer pattern.
 * For high-frequency single-row updates, prefer the targeted helpers
 * (`answerFollowUp`, etc).

export async function saveState(
  notebookId: string,
  userId: string,
  state: JournalState,

  // Replace strategy: delete + insert in a single round trip via two
  // sequential RPCs would be cleaner; for v1 we do it client-side.
  // RLS guards against cross-tenant damage either way.
  await supabase.from("journal_entries").delete().eq("notebook_id", notebookId);
  await supabase.from("entities").delete().eq("notebook_id", notebookId);
  await supabase.from("relations").delete().eq("notebook_id", notebookId);
  await supabase.from("follow_ups").delete().eq("notebook_id", notebookId);
  await supabase.from("asked_questions").delete().eq("notebook_id", notebookId);

  if (state.entries.length) {
    await supabase.from("journal_entries").insert(
      state.entries.map((e) => ({
        id: e.id, user_id: userId, notebook_id: notebookId,
        text: e.text, created_at: e.createdAt,
      })),
    );
  }
  const entityRows = Object.values(state.entities);
  if (entityRows.length) {
    await supabase.from("entities").insert(
      entityRows.map((e) => ({
        id: e.id, user_id: userId, notebook_id: notebookId,
        name: e.name, type: e.type, description: e.description ?? null,
        metadata: e.metadata, mentions: e.mentions,
        first_seen: e.firstSeen, last_seen: e.lastSeen,
        questions: e.questions ?? null,
      })),
    );
  }
  const relationRows = Object.values(state.relations);
  if (relationRows.length) {
    await supabase.from("relations").insert(
      relationRows.map((r) => ({
        id: r.id, user_id: userId, notebook_id: notebookId,
        source_id: r.source, target_id: r.target, label: r.label,
        metadata: r.metadata, created_at: r.createdAt,
        superseded_at: r.supersededAt ?? null,
      })),
    );
  }
  if (state.followUps.length) {
    await supabase.from("follow_ups").insert(
      state.followUps.map((f) => ({
        id: f.id, user_id: userId, notebook_id: notebookId,
        question: f.question, reason: f.reason,
        target_entity_ids: f.targetEntityIds,
        answered: f.answered, created_at: f.createdAt,
      })),
    );
  }
  const asked = state.askedQuestions ?? [];
  if (asked.length) {
    await supabase.from("asked_questions").insert(
      asked.map((q) => ({
        user_id: userId, notebook_id: notebookId,
        question: q.question, target_entity_ids: q.targetEntityIds,
        asked_at: q.askedAt,
      })),
    );
  }
}
```

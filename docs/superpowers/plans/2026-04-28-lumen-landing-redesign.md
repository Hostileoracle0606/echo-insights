# Lumen Landing-Page Redesign Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the static `code.html` landing mockup with a real marketing page mounted at `/`, using the actual app's morphing dock and `d3-force-3d` knowledge graph as decorative animated widgets, while moving the journal experience to `/app`.

**Architecture:** Three coordinated changes. (1) Routing: `src/routes/index.tsx` renames to `app.tsx` (URL becomes `/app`); a new `landing.tsx` claims `/`. (2) Component extraction: the `<DockPill>` inside `dock-preview.tsx` becomes a reusable presentational component under `src/components/landing/`; `<GraphView>` gains an `interactive` prop so the landing graph can disable pan/zoom/click. (3) Three widgets — `EffortlessCaptureWidget` (auto-cycling dock, 12s loop), `GentleInquiryWidget` (animated bubbles, 7s loop, 3 exchange pairs), `CrystallineKnowledgeWidget` (real force graph staging in 10 entities and 9 relations over 15s) — compose those primitives into the landing page. A `<VideoModal>` placeholder backs the "Watch Film" CTA.

**Tech Stack:** TypeScript (strict), React, TanStack Router (file-based routing), Tailwind v4, `d3-force-3d`, `react-force-graph-2d`, Radix Dialog. The project has no automated test harness — validation is `npx tsc --noEmit` plus manual smoke testing via `npm run dev`.

**Companion spec:** [`docs/superpowers/specs/2026-04-28-lumen-landing-redesign-design.md`](../specs/2026-04-28-lumen-landing-redesign-design.md)

## Task 5: Create the `useReducedMotion` hook

**Files:**
- Create: `src/components/landing/use-reduced-motion.ts`

A small hook so each widget can branch on `prefers-reduced-motion` without re-implementing the matchMedia listener.

- [ ] **Step 1: Create the file**

`src/components/landing/use-reduced-motion.ts`:

```ts
import { useEffect, useState } from "react";

/**
 * Tracks the user's `prefers-reduced-motion` setting.
 *
 * Returns `true` when reduce is set, `false` otherwise. SSR-safe — the
 * initial render assumes "no preference" (false), then re-renders if the
 * actual value differs once mounted on the client.
 */
export function useReducedMotion(): boolean {
  const [reduced, setReduced] = useState(false);

  useEffect(() => {
    if (typeof window === "undefined" || !window.matchMedia) return;
    const mql = window.matchMedia("(prefers-reduced-motion: reduce)");
    const update = () => setReduced(mql.matches);
    update();
    mql.addEventListener("change", update);
    return () => mql.removeEventListener("change", update);
  }, []);

  return reduced;
}
```

## Task 8: Build the EffortlessCaptureWidget

**Files:**
- Create: `src/components/landing/effortless-capture-widget.tsx`

Auto-cycling dock with five phases summing to 12 seconds. The user-transcript and agent-transcript reveal word-by-word at ~80 ms per word during their respective phases.

- [ ] **Step 1: Create the file**

`src/components/landing/effortless-capture-widget.tsx`:

```tsx
import { useEffect, useRef, useState } from "react";

import { DockPill, type Mode } from "@/components/landing/dock-pill";
import { DOCK_AGENT_TRANSCRIPT, DOCK_USER_TRANSCRIPT } from "@/components/landing/sample-data";
import { useReducedMotion } from "@/components/landing/use-reduced-motion";

interface Phase {
  mode: Mode;
  duration: number; // ms
}

const PHASES: Phase[] = [
  { mode: "idle", duration: 1000 },
  { mode: "listening", duration: 4500 },
  { mode: "thinking", duration: 1500 },
  { mode: "speaking", duration: 4000 },
  { mode: "idle", duration: 1000 },
];

const WORD_INTERVAL_MS = 80;

function splitWords(text: string): string[] {
  return text.split(/\s+/).filter(Boolean);
}

/**
 * Builds the visible-prefix of a transcript at time `elapsed` (ms) into a
 * phase, capped to the phase duration. ~80 ms per word; the final
 * ~600 ms hold the full transcript so it's readable before the next
 * phase begins.
 */
function transcriptPrefix(text: string, elapsed: number, phaseDuration: number): string {
  const words = splitWords(text);
  if (words.length === 0) return "";
  const reveal = Math.min(elapsed / WORD_INTERVAL_MS, words.length);
  // Hold full text in the last 600 ms of the phase.
  if (elapsed >= phaseDuration - 600) return words.join(" ");
  return words.slice(0, Math.floor(reveal)).join(" ");
}

export function EffortlessCaptureWidget() {
  const reduced = useReducedMotion();
  const [phaseIndex, setPhaseIndex] = useState(0);
  const [phaseElapsed, setPhaseElapsed] = useState(0);
  const startedAtRef = useRef<number>(0);
  const rafRef = useRef<number | null>(null);

  // Phase-advance timer — kicks to the next phase when the current one is up.
  useEffect(() => {
    if (reduced) return;
    startedAtRef.current = performance.now();
    setPhaseElapsed(0);
    const phase = PHASES[phaseIndex];
    const t = window.setTimeout(() => {
      setPhaseIndex((i) => (i + 1) % PHASES.length);
    }, phase.duration);
    return () => window.clearTimeout(t);
  }, [phaseIndex, reduced]);

  // Word-reveal RAF loop — re-renders every animation frame within the phase
  // so transcriptPrefix advances. Stops between phases (no transcript shown
  // during idle/thinking).
  useEffect(() => {
    if (reduced) return;
    const phase = PHASES[phaseIndex];
    if (phase.mode !== "listening" && phase.mode !== "speaking") {
      setPhaseElapsed(phase.duration); // doesn't matter; no transcript reads it
      return;
    }
    const tick = () => {
      const elapsed = performance.now() - startedAtRef.current;
      setPhaseElapsed(elapsed);
      if (elapsed < phase.duration) {
        rafRef.current = requestAnimationFrame(tick);
      }
    };
    rafRef.current = requestAnimationFrame(tick);
    return () => {
      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current);
    };
  }, [phaseIndex, reduced]);

  // Reduced-motion fallback: render the speaking frame statically with the
  // full agent line. Same DOM nesting as the live path so the surrounding
  // card layout is unchanged.
  if (reduced) {
    return (
      <DockPill
        mode="speaking"
        isExpanded
        userText=""
        agentText={DOCK_AGENT_TRANSCRIPT}
      />
    );
  }

  const phase = PHASES[phaseIndex];
  const isExpanded = phase.mode !== "idle";
  const userText =
    phase.mode === "listening"
      ? transcriptPrefix(DOCK_USER_TRANSCRIPT, phaseElapsed, phase.duration)
      : phase.mode === "thinking"
        ? DOCK_USER_TRANSCRIPT
        : "";
  const agentText =
    phase.mode === "speaking"
      ? transcriptPrefix(DOCK_AGENT_TRANSCRIPT, phaseElapsed, phase.duration)
      : "";

  return (
    <DockPill
      mode={phase.mode}
      isExpanded={isExpanded}
      userText={userText}
      agentText={agentText}
    />
  );
}
```

- Create: `src/components/landing/gentle-inquiry-widget.tsx`

- [ ] **Step 1: Create the file**

`src/components/landing/gentle-inquiry-widget.tsx`:

```tsx
import { useEffect, useState } from "react";

import { BUBBLE_EXCHANGES } from "@/components/landing/sample-data";
import { useReducedMotion } from "@/components/landing/use-reduced-motion";
import { cn } from "@/lib/utils";

type Stage = "empty" | "user" | "dots" | "agent" | "fading";

interface StageStep {
  stage: Stage;
  /** Time at which we transition INTO this stage, measured from the start of the loop. */
  at: number;
}

const SCHEDULE: StageStep[] = [
  { stage: "empty", at: 0 },
  { stage: "user", at: 300 },
  { stage: "dots", at: 1500 },
  { stage: "agent", at: 3000 },
  { stage: "fading", at: 6000 },
];
const LOOP_MS = 7000;

function stageAt(elapsed: number): Stage {
  let current: Stage = "empty";
  for (const step of SCHEDULE) {
    if (elapsed >= step.at) current = step.stage;
  }
  return current;
}

export function GentleInquiryWidget() {
  const reduced = useReducedMotion();
  const [exchangeIndex, setExchangeIndex] = useState(0);
  const [stage, setStage] = useState<Stage>("empty");

  useEffect(() => {
    if (reduced) return;
    const startedAt = performance.now();
    let raf = 0;
    const tick = () => {
      const elapsed = (performance.now() - startedAt) % LOOP_MS;
      setStage(stageAt(elapsed));
      raf = requestAnimationFrame(tick);
    };
    raf = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(raf);
  }, [reduced]);

  // When the loop wraps from "fading" back to "empty", advance to the next pair.
  useEffect(() => {
    if (reduced) return;
    if (stage === "empty") {
      setExchangeIndex((i) => (i + 1) % BUBBLE_EXCHANGES.length);
    }
    // No deps on exchangeIndex on purpose — this fires every time we re-enter
    // the empty stage, which happens once per loop.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [stage, reduced]);

  const reducedExchange = BUBBLE_EXCHANGES[0];
  if (reduced) {
    return (
      <div className="space-y-3">
        <Bubble role="user" visible>
          {reducedExchange.user}
        </Bubble>
        <Bubble role="agent" visible>
          {reducedExchange.agent}
        </Bubble>
      </div>
    );
  }

  const exchange = BUBBLE_EXCHANGES[exchangeIndex];
  const userVisible = stage === "user" || stage === "dots" || stage === "agent";
  const agentVisible = stage === "agent";
  const dotsVisible = stage === "dots";
  const fading = stage === "fading";

  return (
    <div className={cn("space-y-3 transition-opacity duration-200", fading && "opacity-0")}>
      <Bubble role="user" visible={userVisible}>
        {exchange.user}
      </Bubble>
      {dotsVisible ? (
        <Bubble role="agent" visible>
          <Dots />
        </Bubble>
      ) : (
        <Bubble role="agent" visible={agentVisible}>
          {exchange.agent}
        </Bubble>
      )}
    </div>
  );
}

function Bubble({
  role,
  visible,
  children,
}: {
  role: "user" | "agent";
  visible: boolean;
  children: React.ReactNode;
}) {
  return (
    <div
      className={cn(
        "rounded-lg p-3 text-sm leading-snug shadow-sm transition-all duration-300",
        role === "user"
          ? "ml-8 rounded-tr-none bg-white text-[#4d4635]"
          : "mr-8 rounded-tl-none border border-[#d5e0f7]/60 bg-[#d5e0f7]/30 text-[#1a1c1a]",
        visible ? "translate-y-0 opacity-100" : "translate-y-2 opacity-0",
      )}
    >
      {children}
    </div>
  );
}

function Dots() {
  return (
    <span className="inline-flex items-center gap-1.5 py-0.5">
      {[0, 1, 2].map((i) => (
        <span
          key={i}
          className="h-1.5 w-1.5 rounded-full bg-[#4d4635]"
          style={{ animation: `landing-dot-pulse 1.4s ease-in-out ${i * 0.15}s infinite` }}
        />
      ))}
    </span>
  );
}
```

## Task 10: Build the LandingGraphLoop controller

**Files:**
- Create: `src/components/landing/landing-graph-loop.tsx`

Wraps `<GraphView>` and feeds it a growing slice of `LANDING_ENTITIES` / `LANDING_RELATIONS` over an 8-second stage-in, then a 6-second drift, then a 1-second fade, then loop reset (15 s total).

- [ ] **Step 1: Create the file**

`src/components/landing/landing-graph-loop.tsx`:

```tsx
import { useEffect, useState } from "react";

import { GraphView } from "@/components/graph-view";
import { LANDING_ENTITIES, LANDING_RELATIONS } from "@/components/landing/sample-data";
import { useReducedMotion } from "@/components/landing/use-reduced-motion";
import { cn } from "@/lib/utils";

const STAGE_IN_MS = 8000;
const DRIFT_MS = 6000;
const FADE_MS = 1000;
const LOOP_MS = STAGE_IN_MS + DRIFT_MS + FADE_MS;
const ENTITY_INTERVAL_MS = STAGE_IN_MS / LANDING_ENTITIES.length; // ~800 ms per entity

interface Props {
  className?: string;
}

export function LandingGraphLoop({ className }: Props) {
  const reduced = useReducedMotion();
  const [visibleCount, setVisibleCount] = useState<number>(reduced ? LANDING_ENTITIES.length : 0);
  const [opacity, setOpacity] = useState(1);

  useEffect(() => {
    if (reduced) {
      setVisibleCount(LANDING_ENTITIES.length);
      setOpacity(1);
      return;
    }
    let raf = 0;
    const startedAt = performance.now();
    const tick = () => {
      const elapsed = (performance.now() - startedAt) % LOOP_MS;

      if (elapsed < STAGE_IN_MS) {
        const count = Math.min(
          LANDING_ENTITIES.length,
          Math.floor(elapsed / ENTITY_INTERVAL_MS) + 1,
        );
        setVisibleCount(count);
        setOpacity(1);
      } else if (elapsed < STAGE_IN_MS + DRIFT_MS) {
        setVisibleCount(LANDING_ENTITIES.length);
        setOpacity(1);
      } else {
        // Fade phase.
        const fadeElapsed = elapsed - (STAGE_IN_MS + DRIFT_MS);
        setVisibleCount(LANDING_ENTITIES.length);
        setOpacity(Math.max(0, 1 - fadeElapsed / FADE_MS));
      }

      raf = requestAnimationFrame(tick);
    };
    raf = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(raf);
  }, [reduced]);

  // Slice entities by `visibleCount`. Filter relations to those whose source
  // AND target are both in the visible slice — otherwise GraphView's
  // entity-membership filter on `data.links` drops them anyway, but slicing
  // here is cheap and keeps the visible state explicit.
  const visibleEntities = LANDING_ENTITIES.slice(0, visibleCount);
  const visibleIds = new Set(visibleEntities.map((e) => e.id));
  const visibleRelations = LANDING_RELATIONS.filter(
    (r) => visibleIds.has(r.source) && visibleIds.has(r.target),
  );

  return (
    <div
      className={cn("h-full w-full transition-opacity", className)}
      style={{ opacity }}
    >
      <GraphView
        entities={visibleEntities as Parameters<typeof GraphView>[0]["entities"]}
        relations={visibleRelations as Parameters<typeof GraphView>[0]["relations"]}
        interactive={false}
        className="h-full w-full bg-transparent"
      />
    </div>
  );
}
```

(The `as Parameters<typeof GraphView>[0]["entities"]` casts strip the `ReadonlyArray` covariance — the underlying GraphView mutates internally via `d3-force-3d`, but our seed data is shared across renders, so passing readonly arrays is safe at runtime; the cast just satisfies TypeScript.)
